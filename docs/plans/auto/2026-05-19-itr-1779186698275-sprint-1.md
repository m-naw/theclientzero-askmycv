async function runFetch(request: Request): Promise<Response> {

  await waitOnExecutionContext(ctx);
  return res;

async function drainStream(res: Response): Promise<void> {
  if (!res.body) return;
  const reader = res.body.getReader();
  for (;;) {
    const { done } = await reader.read();
    if (done) break;
  }
}

describe("KV-put failure — spend tracking must not throw, chat must still return 200", () => {

    const putSpy = vi.spyOn(stateKv, "put").mockImplementation(async (key: string, value: string, options?: Parameters<KVNamespace["put"]>[2]) => {

        method: "POST",
        headers: {

          "user-agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",

      putSpy.mockRestore();
